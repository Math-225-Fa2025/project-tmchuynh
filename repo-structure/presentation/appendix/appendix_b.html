# Function to retrieve rating history for a given user
get_rating_history <- function(username) {
    cat(paste0("\nFetching data for user: ", username, "\n"))
    Sys.sleep(runif(1, 0.5, 1.2)) # polite delay
    url <- paste0("https://lichess.org/api/user/", username, "/rating-history")
    tryCatch(
        {
            response <- fromJSON(url, simplifyDataFrame = FALSE)
            cat(paste0("  Response length: ", length(response), "\n"))
            if (length(response) == 0) {
                cat("  No rating history available\n")
                return(NULL)
            }
            
            # Process each game type
            result_list <- list()
            for (i in seq_along(response)) {
                item <- response[[i]]
                game_type <- item[["name"]]
                points_data <- item[["points"]]
                
                if (!is.null(points_data) && length(points_data) > 0) {
                    # Handle different possible structures
                    if (is.list(points_data) && !is.data.frame(points_data)) {
                        # Convert list of vectors to matrix
                        points_matrix <- matrix(unlist(points_data), ncol = 2, byrow = TRUE)
                    } else if (is.matrix(points_data)) {
                        points_matrix <- points_data
                    } else {
                        next  # Skip if unexpected format
                    }
                    
                    result_list[[i]] <- tibble(
                        user = username,
                        category = game_type,
                        date_index = points_matrix[, 1],
                        rating = points_matrix[, 2]
                    ) %>%
                        mutate(date = as_date("2013-01-01") + weeks(date_index))
                }
            }
            
            if (length(result_list) > 0) {
                return(bind_rows(result_list))
            } else {
                cat("  No valid rating points found\n")
                return(NULL)
            }
        },
        error = function(e) {
            cat(paste0("  Error: ", e$message, "\n"))
            return(NULL)
        }
    )
}

B.2 Time-Series Cleaning and Aggregation

# Clean, filter, and aggregate by category (e.g., Blitz, Rapid)
rating_summary <- rating_histories %>%
    filter(!is.na(rating)) %>%
    group_by(user, category) %>%
    mutate(
        rating_smooth = zoo::rollapply(rating, width = 3, FUN = mean, fill = NA, align = "right"),
        relative_time = row_number()
    ) %>%
    ungroup()

Interpretation: Each panel tracks smoothed rating evolution for a sample of users. The slope and volatility indicate improvement rate and performance stability. Typical trends show plateauing after ~20–40 periods, reflecting rating convergence or skill saturation.

B.4 Analytical Highlights

  • Consistent upward trends imply sustained learning and adaptation to faster time controls.
  • Volatile fluctuations correspond to periods of experimentation or inactivity.
  • Cross-category gaps (e.g., Blitz > Rapid > Classical) align with expected behavioral rating hierarchies in online chess.
  • Aggregating trajectories could enable a growth-rate model (e.g., mixed-effects regression) quantifying individual improvement rates.
